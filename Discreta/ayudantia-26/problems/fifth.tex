\begin{problem}{}{}

    Calcular, evaluando, las siguientes expresiones:

    \begin{enumerate}[(a)]
        \item $\sum_{r=0}^{4} r$
        \item $\prod_{i=1}^{5} i$
        \item $\sum_{k=-3}^{-1} \frac{1}{k \cdot (k+4)}$
        \item $\prod_{n=2}^{7} \frac{n}{n-1}$
    \end{enumerate}

\end{problem}
\vspace{1ex}

A continuación están las evaluaciones completas de las expresiones:

\begin{enumerate}[(a)]
    \item \begin{tabular}{|l l r}
         & $\sum_{r=0}^{4} r$ & \\
        $=$ & $0 + 1 + 2 + 3 + 4$ & (expansión de $\Sigma$) \\
        $=$ & $10$ & (aritmética) \\
    \end{tabular}

    \item \begin{tabular}{|l l r}
         & $\prod_{i=1}^{5} i$ & \\
        $=$ & $1 \cdot 2 \cdot 3 \cdot 4 \cdot 5$ & (expansión de $\Pi$) \\
        $=$ & $120$ & (aritmética) \\
    \end{tabular}

    \item \begin{tabular}{|l l r}
         & $\sum_{k=-3}^{-1} \frac{1}{k \cdot (k+4)}$ & \\
        $=$ & $\frac{1}{(-3)(-3 + 4)} + \frac{1}{(-2)(-2 + 4)} + \frac{1}{(-1)(-1 + 4)}$ & (expansión de $\Sigma$) \\
        $=$ & $-\frac{1}{3} - \frac{1}{4} - \frac{1}{3}$ & (aritmética) \\
        $=$ & $-\frac{11}{12}$ & (aritmética) \\
    \end{tabular}

    \item \begin{tabular}{|l l r}
         & $\prod_{n=2}^{7} \frac{n}{n-1}$ & \\
        $=$ & $\frac21 \cdot \frac32 \cdot \frac43 \cdot \frac54 \cdot \frac65 \cdot \frac76$ & (expansión de $\Pi$) \\
        $=$ & $7$ & (producto telescópico) \\
    \end{tabular}
\end{enumerate}
